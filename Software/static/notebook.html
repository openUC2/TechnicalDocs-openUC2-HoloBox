# Core hologram processing functions
def abssqr(x):
    """Calculate intensity (what a detector sees)"""
    return np.real(x * np.conj(x))

def FT(x):
    """Forward Fourier transform with proper frequency shift"""
    return np.fft.fftshift(np.fft.fft2(x))

def iFT(x):
    """Inverse Fourier transform with proper frequency shift"""
    return np.fft.ifft2(np.fft.ifftshift(x))

def fresnel_propagator(E0, ps, lambda0, z):
    """
    Freespace propagation using Fresnel kernel
    
    Args:
        E0: Initial complex field in x-y source plane
        ps: Pixel size in meters
        lambda0: Wavelength in meters  
        z: Distance from sensor to object in meters
    
    Returns:
        Ef: Propagated output field
    """
    upsample_scale = 1
    n = upsample_scale * E0.shape[1]  # Image width in pixels
    grid_size = ps * n                # Grid size in x-direction
    
    # Inverse space (frequency domain)
    fx = np.linspace(-(n-1)/2*(1/grid_size), (n-1)/2*(1/grid_size), n)
    fy = np.linspace(-(n-1)/2*(1/grid_size), (n-1)/2*(1/grid_size), n)
    Fx, Fy = np.meshgrid(fx, fy)
    
    # Fresnel kernel / point spread function
    H = np.exp(1j*(2 * np.pi / lambda0) * z) * np.exp(1j * np.pi * lambda0 * z * (Fx**2 + Fy**2))
    
    # Apply Fresnel propagation
    Ef = iFT(FT(E0) * H)
    
    return Ef

# Demo hologram processing
def create_sample_hologram(size=256):
    """Create a sample hologram for demonstration"""
    # Create a synthetic hologram pattern
    x = np.linspace(-1, 1, size)
    y = np.linspace(-1, 1, size)
    X, Y = np.meshgrid(x, y)
    
    # Create some sample objects (circles)
    obj1 = np.exp(-((X-0.3)**2 + (Y-0.2)**2) / 0.01)
    obj2 = np.exp(-((X+0.2)**2 + (Y-0.3)**2) / 0.02)
    
    # Add some interference patterns
    hologram = 1 + 0.5 * np.cos(10 * np.pi * X) * obj1 + 0.3 * np.cos(15 * np.pi * Y) * obj2
    
    # Add some noise
    hologram += 0.1 * np.random.random((size, size))
    
    return hologram

def process_and_display_hologram():
    """Process hologram with current parameters and display results"""
    # Get current parameters
    wavelength = float(document.getElementById('wavelength').value) * 1e-9  # nm to m
    pixelsize = float(document.getElementById('pixelsize').value) * 1e-6   # µm to m
    distance = float(document.getElementById('distance').value) * 1e-3     # mm to m
    
    # Create sample hologram
    hologram = create_sample_hologram()
    
    # Process hologram
    reconstructed = fresnel_propagator(hologram.astype(complex), pixelsize, wavelength, distance)
    
    # Calculate intensity
    intensity = abssqr(reconstructed)
    
    # Display results (simplified for demo)
    print(f"Processed hologram with:")
    print(f"  Wavelength: {wavelength*1e9:.0f} nm")
    print(f"  Pixel size: {pixelsize*1e6:.1f} µm") 
    print(f"  Distance: {distance*1e3:.1f} mm")
    print(f"  Result shape: {intensity.shape}")
    print(f"  Intensity range: {intensity.min():.3f} - {intensity.max():.3f}")
    
    return hologram, intensity
